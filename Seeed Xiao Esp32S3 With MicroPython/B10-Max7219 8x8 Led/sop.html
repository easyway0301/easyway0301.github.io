MAX7219 8x8 LED

第一步：模組介紹

本教學使用 MAX7219 8x8 LED 矩陣模組，透過 SPI 控制顯示文字。

模組特點：
• 8x8 LED 矩陣
• SPI 通訊控制
• 可串接多個模組

接線方式 (Seeed XIAO ESP32-S3)：

  • VCC → 3V3
  • GND → GND
  • DIN / MOSI → GPIO 9
  • SCK → GPIO 7
  • CS → GPIO 8
圖片來源： Seeed Studio 官方文件

第二步：MicroPython 程式碼

以下程式碼可直接在 Thonny 上執行，顯示跑馬燈文字 "HELLO"。

max72198x8led.py


from machine import Pin, SPI      # 匯入 SPI 與 Pin 類別
import max7219                    # 匯入 MAX7219 模組
import time                        # 匯入時間模組

# === 初始化 SPI ===
spi = SPI(1, baudrate=10_000_000, sck=Pin(7), mosi=Pin(9))   # SPI 物件，指定 SCK 與 MOSI
cs = Pin(8, Pin.OUT)                                         # CS 腳位

# === 初始化 MAX7219 ===
display = max7219.Matrix8x8(spi, cs, 1)                      # 1 個 8x8 模組
display.brightness(5)                                        # 設定亮度 0~15

# === 顯示跑馬燈文字 ===
# 往左移動文字從右邊開始跑到左邊
for x in range(32, -40, -1):                                 # 文字 x 座標從 32 到 -40
    display.fill(0)                                          # 清空畫面
    display.text("HELLO", x, 0, 1)                           # 顯示文字
    display.show()                                           # 更新顯示
    time.sleep(0.1)                                         # 延遲 0.1 秒

max7219.py


# max7219.py
from micropython import const
import framebuf

# MAX7219 registers
_DIGIT0 = const(0x01)
_DECODEMODE = const(0x09)
_INTENSITY = const(0x0A)
_SCANLIMIT = const(0x0B)
_SHUTDOWN = const(0x0C)
_DISPLAYTEST = const(0x0F)

class Matrix8x8(framebuf.FrameBuffer):
    def __init__(self, spi, cs, num):
        self.spi = spi
        self.cs = cs
        self.cs.init(self.cs.OUT, value=1)
        self.num = num
        self.buffer = bytearray(8 * num)
        super().__init__(self.buffer, 8 * num, 8, framebuf.MONO_HLSB)
        self.init()

    def init(self):
        for cmd, data in (
            (_DISPLAYTEST, 0),
            (_SCANLIMIT, 7),
            (_DECODEMODE, 0),
            (_SHUTDOWN, 1),
        ):
            self._write(cmd, data)
        self.brightness(0)

    def brightness(self, value):
        if not 0 <= value <= 15:
            raise ValueError("Brightness out of range")
        self._write(_INTENSITY, value)

    def _write(self, command, data):
        self.cs(0)
        for _ in range(self.num):
            self.spi.write(bytearray([command, data]))
        self.cs(1)

    def show(self):
        for i in range(8):
            self.cs(0)
            for j in range(self.num):
                self.spi.write(
                    bytearray([_DIGIT0 + i, self.buffer[(j * 8) + i]])
                )
            self.cs(1)

執行結果

  • 文字 "HELLO" 從右往左平滑滾動
  • 亮度可透過display.brightness()調整
  • 可延伸成跑馬燈選單、訊息顯示
